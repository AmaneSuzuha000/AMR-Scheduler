cmake_minimum_required(VERSION 3.16)
project(amr_gazebo_guard)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(gazebo_dev REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(gazebo REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
find_package(nav2_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(dwb_core REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/MotionLease.msg" "msg/GuardStatus.msg" "srv/AuthorizeMotion.srv"
  DEPENDENCIES std_msgs)
add_library(amr_motion_guard SHARED src/motion_guard.cpp)
target_include_directories(amr_motion_guard PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(amr_motion_guard ${GAZEBO_LIBRARIES})
ament_target_dependencies(amr_motion_guard rclcpp gazebo_ros geometry_msgs std_msgs)
rosidl_get_typesupport_target(guard_typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(amr_motion_guard "${guard_typesupport}")
target_compile_options(amr_motion_guard PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS amr_motion_guard LIBRARY DESTINATION lib)
add_library(amr_bounded_pursuit SHARED src/bounded_pursuit.cpp)
ament_target_dependencies(amr_bounded_pursuit nav2_regulated_pure_pursuit_controller nav2_core pluginlib rclcpp)
target_compile_options(amr_bounded_pursuit PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(nav2_core bounded_pursuit.xml)
install(TARGETS amr_bounded_pursuit LIBRARY DESTINATION lib)
add_library(amr_continuous_goal SHARED src/continuous_goal.cpp)
ament_target_dependencies(amr_continuous_goal dwb_core pluginlib)
target_compile_options(amr_continuous_goal PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(dwb_core continuous_goal.xml)
install(TARGETS amr_continuous_goal LIBRARY DESTINATION lib)
add_executable(check_continuous_goal test/check_continuous_goal.cpp)
ament_target_dependencies(check_continuous_goal dwb_core pluginlib)
install(TARGETS check_continuous_goal DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_twist_bounds test/test_twist_bounds.cpp)
  add_test(NAME execution_twist_bounds COMMAND test_twist_bounds)
  add_executable(test_motion_watchdog test/test_motion_watchdog.cpp)
  add_test(NAME execution_motion_watchdog COMMAND test_motion_watchdog)
endif()
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
